// --- Mock Data ---

const mockCourse = {
  name: "Augusta National",
  hole: 12,
  par: 3,
  holeName: "Golden Bell",
  mapUrl: "https://placehold.co/600x400/2C5E1A/FFFFFF?text=Hole+12+Map+View",
  tees: {
    masters: { name: "Masters", yardage: 155, green: { front: 145, middle: 155, back: 165 } },
    members: { name: "Members", yardage: 140, green: { front: 130, middle: 140, back: 150 } },
    ladies: { name: "Ladies", yardage: 115, green: { front: 105, middle: 115, back: 125 } },
  },
  advice: "A very dangerous par 3. With water in front and trouble long and left, the center of the green is always the target, regardless of pin position. Club selection is critical as the wind swirls."
};

const defaultUserClubs = {
  'Driver': 250,
  '3-Wood': 225,
  '5-Wood': 205,
  '4-Iron': 190,
  '5-Iron': 180,
  '6-Iron': 170,
  '7-Iron': 160,
  '8-Iron': 150,
  '9-Iron': 140,
  'PW': 130,
  'GW': 115,
  'SW': 95,
};

// --- Main App Component ---

export default function App() {
  const [activeTab, setActiveTab] = useState('caddy');
  const [selectedTee, setSelectedTee] = useState('masters');
  const [userClubs, setUserClubs] = useState(defaultUserClubs);
  const [showSettings, setShowSettings] = useState(false);

  const holeData = mockCourse.tees[selectedTee];

  // --- Logic for Club Recommendation ---
  const recommendedClub = useMemo(() => {
    const targetDistance = holeData.green.middle;
    let bestClub = null;
    let smallestDifference = Infinity;

    for (const [club, distance] of Object.entries(userClubs)) {
      const difference = Math.abs(targetDistance - distance);
      if (difference < smallestDifference) {
        smallestDifference = difference;
        bestClub = club;
      }
    }
    return bestClub;
  }, [selectedTee, userClubs, holeData]);

  return (
    <div className="flex justify-center items-center h-screen bg-gray-900 font-inter">
      {/* Mobile App Shell */}
      <div className="w-full max-w-sm h-full sm:h-[90vh] sm:max-h-[840px] bg-gray-100 rounded-lg shadow-2xl overflow-hidden flex flex-col">
        
        {/* Header */}
        <header className="bg-white shadow-md p-4 flex justify-between items-center z-10">
          <h1 className="text-2xl font-bold text-green-700">TeeBox</h1>
          <button onClick={() => setShowSettings(true)} className="text-gray-600 hover:text-green-700">
            <SettingsIcon className="w-6 h-6" />
          </button>
        </header>

        {/* Main Content Area */}
        <main className="flex-1 overflow-y-auto pb-16">
          {activeTab === 'caddy' && <CaddyView holeData={holeData} selectedTee={selectedTee} setSelectedTee={setSelectedTee} recommendedClub={recommendedClub} />}
          {activeTab === 'coach' && <SwingCoachView />}
        </main>

        {/* Bottom Navigation */}
        <nav className="absolute bottom-0 left-1/2 -translate-x-1/2 w-full max-w-sm h-16 bg-white border-t border-gray-200 flex justify-around items-center">
          <TabButton
            icon={<GolfIcon className="w-7 h-7" />}
            label="Caddy"
            isActive={activeTab === 'caddy'}
            onClick={() => setActiveTab('caddy')}
          />
          <TabButton
            icon={<VideoIcon className="w-7 h-7" />}
            label="Swing Coach"
            isActive={activeTab === 'coach'}
            onClick={() => setActiveTab('coach')}
          />
        </nav>

        {/* Settings Modal */}
        {showSettings && <SettingsModal userClubs={userClubs} setUserClubs={setUserClubs} onClose={() => setShowSettings(false)} />}
      </div>
    </div>
  );
}

// --- Settings Modal Component ---

function SettingsModal({ userClubs, setUserClubs, onClose }) {
  
  const handleClubChange = (club, distance) => {
    // Only allow numbers
    const newDistance = parseInt(distance.replace(/\D/g, ''), 10);
    setUserClubs(prev => ({
      ...prev,
      [club]: isNaN(newDistance) ? 0 : newDistance,
    }));
  };

  return (
    <div className="absolute inset-0 bg-black/50 flex justify-center items-center p-4 z-50">
      <div className="bg-white rounded-lg shadow-2xl w-full max-w-md max-h-[90vh] flex flex-col">
        {/* Modal Header */}
        <div className="flex justify-between items-center p-4 border-b">
          <h2 className="text-xl font-bold text-gray-800">My Club Distances</h2>
          <button onClick={onClose} className="text-gray-500 hover:text-gray-800">
            <XIcon className="w-6 h-6" />
          </button>
        </div>
        
        {/* Modal Body */}
        <div className="p-4 overflow-y-auto space-y-2">
          <p className="text-sm text-gray-600 mb-4">Enter your average carry distance (in yards) for each club. This will power your Caddy recommendations.</p>
          {Object.keys(userClubs).map(club => (
            <div key={club} className="flex items-center space-x-3">
              <label className="w-1/3 font-semibold text-gray-700">{club}</label>
              <input
                type="tel" // Use 'tel' for number pad on mobile
                value={userClubs[club]}
                onChange={(e) => handleClubChange(club, e.target.value)}
                className="flex-1 p-2 border border-gray-300 rounded-md text-right"
              />
              <span className="text-gray-500">yds</span>
            </div>
          ))}
        </div>
        
        {/* Modal Footer */}
        <div className="p-4 border-t bg-gray-50">
          <button 
            onClick={onClose} 
            className="w-full py-2 px-4 bg-green-700 text-white font-bold rounded-lg shadow hover:bg-green-800 transition-all"
          >
            Save & Close
          </button>
        </div>
      </div>
    </div>
  );
}
